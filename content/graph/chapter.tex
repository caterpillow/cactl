\subsection{Theorems}

  	A matching $M$ is maximum \textbf{iff} there is no $M$-augmenting path. \\

	\subsection{Hall's Marriage Theorem}
		$G = (L, R; E)$ has a matching covering all of $L$ iff $|N(S)| \ge |S|$ for every $S \subseteq L$. \\
		Corollary: Let $\delta=\max_{S\subseteq L}\big(|S|-|N(S)|\big)$. The size of the maximum matching is $\nu = |L|-\delta$. \\

	\subsection{Kőnig's Theorem}
 		In bipartite graphs: \textbf{max matching = min vertex cover}, i.e. $\nu(G)=\tau(G)$.
		Hence max independent set size $\alpha(G)=|V|-\tau(G)=|V|-\nu(G)$. \\

		To recover min vertex cover: 
		From all unmatched vertices in $L$, BFS alternating paths (use non-matching edges $L\!\to\!R$; matching edges $R\!\to\!L$).
		Let $Z$ be reachable vertices. Min vertex cover is $(L\setminus Z)\cup(R\cap Z)$. \\

	\subsection{Dilworth's Theorem (posets)}
		Run Floyd-Warshall for transitivity. \\
		Width (max antichain size) = min number of chains in a partition. \\
		Reduction: make bipartite graph with two copies of elements; edge $x_L\!-\!y_R$ if $x<y$. Then
		width $=$ min chain decomposition size $= |P|-\nu$. \\
		Mirsky's Theorem (dual): Height (max chain size) = min number of antichains in a partition. \\
		Max antichain: from a min vertex cover, take the $x$ with neither $x_L$ nor $x_R$ in it. \\
		Min vertex-disjoint path cover of a DAG $=|V|-\nu$, with an edge $x_L\!-\!y_R$ per DAG edge. \\

	\subsection{Tutte's 1-Factor Theorem}
$\text{odd}(H)$ is the number of components of a graph $H$ with an odd number of vertices.
		$G$ has a perfect matching iff for all $S\subseteq V$, $\text{odd}(G-S)\le |S|$.
		Tutte--Berge formula: $\displaystyle \nu(G)=\tfrac12\min_{S\subseteq V}\big(|V|-\text{odd}(G-S)+|S|\big)$. \\

	\subsection{Edge cover}
		In any graph without isolated vertices, the size of the minimum edge cover (set of edges that touch every vertex) is given by: \[\rho'(G)=|V|-\nu(G)\] \\

\section{Math}
	\subsection{Number of Spanning Trees}
		Create an $N\times N$ matrix \texttt{mat}, and for each edge $a \rightarrow b \in G$, do
		\texttt{mat[a][b]---, mat[b][b]++} (and \texttt{mat[b][a]---, mat[a][a]++} if $G$ is undirected).
		Remove the $i$th row and column and take the determinant; this yields the number of directed spanning trees rooted at $i$ with edges pointing away from $i$
		(if $G$ is undirected, remove any row/column).
		% I.e. matrix-tree theorem.
		% Source: https://en.wikipedia.org/wiki/Kirchhoff%27s_theorem
		% Test: stress-tests/graph/matrix-tree.cpp

	\subsection{Erdős-Gallai theorem}
		A simple graph with node degrees $d_1 \ge \dots \ge d_n$ exists iff $d_1 + \dots + d_n$ is even and for every $k = 1\dots n$,
		\[ \sum _{i=1}^{k}d_{i}\leq k(k-1)+\sum _{i=k+1}^{n}\min(d_{i},k). \]
		% Source: https://en.wikipedia.org/wiki/Erd%C5%91s%E2%80%93Gallai_theorem
		% Test: stress-tests/graph/erdos-gallai.cpp

	\subsection{Euler's formula (planar graphs)}
		A connected planar graph has $V - E + F = 2$ (counting the outer face); with $C$ components, $V - E + F = 1 + C$.
		Simple planar with $V \ge 3$: $E \le 3V - 6$ (and $E \le 2V - 4$ if triangle-free), so average degree $< 6$.
		% Source: https://en.wikipedia.org/wiki/Euler_characteristic#Plane_graphs

	\subsection{BEST theorem}
	Connected digraph with indeg $=$ outdeg $=\deg(v)$ everywhere: the number of Eulerian circuits (cyclic sequences of distinct edges) is $t_w(G)\prod_v(\deg(v)-1)!$, where $t_w$ is the number of spanning arborescences rooted at $w$ (any $w$; the matrix of Number of Spanning Trees). Euler trails $s\to t$: add an arc $t\to s$ first.

	\subsection{Reductions}
	
	\subsubsection{Min cut reduction}
		Consider min cut as an instance of this problem: 
		Given $A[N]$ and $B[N]$ (any sign) and $C[N][N]$ (non-negative), find a binary string $S$ of length $N$ that minimises the following: \\
		\begin{itemize}[noitemsep]
		\item Pay $A[i]$ if $S[i] = 1$ 
		\item Pay $B[i]$ if $S[i] = 0$ 
		\item Pay $C[i][j]$ if $S[i] = 1$ and $S[j] = 0$. 
		\end{itemize}
		To calculate the answer ($S[i] = 1$ means $i$ is on the $s$ side): 
		\begin{itemize}[noitemsep]
		\item Add edges $s$ to $i$ of capacity $B[i] + \texttt{inf}$. 
		\item Add edges $i$ to $t$ of capacity $A[i] + \texttt{inf}$. 
		\item Add edges $i$ to $j$ of capacity $C[i][j]$. 
\item The answer is the maximum flow in this graph minus $N \times \texttt{inf}$, where $\texttt{inf} \ge \max_i(-A[i], -B[i])$.
		\end{itemize}

	\subsection{Circulation \& Lower Bounds}

	Each edge $e=(u\to v)$ has bounds $[\ell_e, u_e]$.
	Find flows $f_e$ such that:
	\[
	\ell_e \le f_e \le u_e,\qquad
	\sum f_{\text{in}} = \sum f_{\text{out}} \text{ at every node.}
	\]
	Let $f'_e = f_e - \ell_e$, so $0 \le f'_e \le u_e - \ell_e$.
	Define node demand:
	\[
	d[v] = \sum_{e=(v\to w)} \ell_e - \sum_{e=(u\to v)} \ell_e
	\]
	Then residual flow must satisfy
	\[
	\sum f'_{\text{in}} - \sum f'_{\text{out}} = d[v].
	\]

	Build auxiliary graph:

	\begin{itemize}[noitemsep]
	\item For each edge $(u\to v)$, add cap $u_e-\ell_e$
	\item Add super source $S$, super sink $T$
	\item If $d[v]>0$: add $v\to T$ cap $d[v]$
	\item If $d[v]<0$: add $S\to v$ cap $-d[v]$
	\end{itemize}

	Let $D=\sum_{d[v]>0} d[v]$. Feasible circulation iff max $S \rightarrow T$ flow equals $D$.

	To recover original flow, use $f_e = f'_e + \ell_e$.
	We can model max $s \rightarrow t$ flow with lower bounds by adding edge $(t, s)$ with bounds $[L, \infty)$ and binary searching the largest feasible $L$.

	\subsection{Tree facts}
	For vertex sets $A,B$ with diameter endpoints $(a_1,a_2),(b_1,b_2)$, the diameter of $A\cup B$ has endpoints among those four. The farthest vertex from any vertex is a diameter endpoint. Centre = middle of a diameter (one vertex if its length $D$ is even, else two adjacent), radius $\lceil D/2\rceil$.
	Centroid: every component after removal has size $\le n/2$; one or two (adjacent).
